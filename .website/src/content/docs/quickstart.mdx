This quickstart will walk you through how to get everything setup to run
a basic producer and consumer using SQLStreams.

If you just want the end result, clone [this](TODO create github repo example).

## 0. Prerequisites

- Go 1.27+
- Docker Compose for running Postgres

## 1. Prepare the workspace

Create a directory and install the library:

```sh
mkdir sqlstreams-quickstart
cd sqlstreams-quickstart
go mod init example.com/sqlstreams-quickstart
go get github.com/allegedlyreliable/sqlstreams@v0.1.1
mkdir -p cmd/produce cmd/consume
```

Next lets create our docker compose file for running Postgres.

```sh
touch compose.yaml
```

```yaml
# compose.yaml
name: quickstart

services:
  db:
    image: postgres:18-alpine
    environment:
      POSTGRES_USER: username
      POSTGRES_PASSWORD: password
      POSTGRES_DB: quickstart
    volumes:
      - pgdata:/var/lib/postgresql/data
    ports:
      - "5432:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U username -d quickstart"]
      interval: 10s
      timeout: 5s
      retries: 5

volumes:
  pgdata:
```

Spin it up

```sh
docker compose up -d
```

## 2. Produce a message

Save this as `cmd/produce/main.go`:

```go
// cmd/produce/main.go
package main

import (
	"fmt"
	"os"

	"github.com/allegedlyreliable/sqlstreams/client"
)

type WelcomeEmail struct {
	UserId string `json:"user_id"`
}

func (WelcomeEmail) SchemaVersion() int { return 1 } // increment on breaking changes

func main() {
	if err := run(); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

func run() error {
	ctx, stop := sqlstreams.LifecycleContext(nil)
	defer stop()

	pool, err := sqlstreams.NewPostgresPool(ctx, "username", "password", "localhost", "quickstart", nil)
	if err != nil {
		return err
	}
	defer pool.Close()

	client, err := sqlstreams.NewClient(ctx, pool, nil)
	if err != nil {
		return err
	}

	emails := client.Stream[WelcomeEmail]("signup.welcome-email")
	if _, err := emails.Register(ctx, nil); err != nil {
		return err
	}

	producer, err := emails.Producer().Register(ctx, nil)
	if err != nil {
		return err
	}

	produced, err := producer.Produce(ctx, &WelcomeEmail{UserId: "user-123"}, nil)
	if err != nil {
		return err
	}
	fmt.Printf("produced message id=%d\n", produced.Id)
	return nil
}
```

<ThreadAside label="NOTE" title="Import Issues">
If you are getting an import error for SQLStreams you
may need to run `go mod tidy`.
</ThreadAside>

Run it:

```sh
go run ./cmd/produce
```

The output should look something like this:

```text
produced message id=1
```

You can run this as many times as you'd like and the message id will increment with each run.

## 3. Consume the message

Save this as `cmd/consume/main.go`:

```go
// cmd/consume/main.go
package main

import (
	"context"
	"errors"
	"fmt"
	"os"

	"github.com/allegedlyreliable/sqlstreams/client"
)

type WelcomeEmail struct {
	UserId string `json:"user_id"`
}

func (WelcomeEmail) SchemaVersion() int { return 1 } // increment on breaking changes

func main() {
	if err := run(); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

func run() error {
	ctx, stop := sqlstreams.LifecycleContext(nil)
	defer stop()

	pool, err := sqlstreams.NewPostgresPool(ctx, "username", "password", "localhost", "quickstart", nil)
	if err != nil {
		return err
	}
	defer pool.Close()

	client, err := sqlstreams.NewClient(ctx, pool, nil)
	if err != nil {
		return err
	}

	emails := client.Stream[WelcomeEmail]("signup.welcome-email")
	registered, err := emails.Get(ctx)
	if err != nil {
		return err
	}
	if registered == nil {
		return errors.New("stream not registered -- run the producer first")
	}

	sender := emails.Consumer("email-sender")
	consumer, err := sender.Register(ctx, nil)
	if err != nil {
		return err
	}

	return consumer.Consume(ctx, receiveEmail, nil)
}

func receiveEmail(ctx context.Context, email *WelcomeEmail) error {
	fmt.Printf("received welcome email request for %s\n", email.UserId)
	return nil
}
```

Run it:

```sh
go run ./cmd/consume
```

The output should look something like this:

```text
received welcome email request for user-123
```

<ThreadAside label="NOTE" title="Blocking Consumer">
The consumer is blocking and will keep waiting for messages. 
Press Ctrl-C to stop it.
</ThreadAside>

And thats it. You can have a seperate terminal session run the 
producer and watch the consumer `consume` the `produced` messages in
real time.
